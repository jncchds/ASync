// Measures a large TwoWay upload through the gateway: time, server memory and server temp disk, and what an
// interrupted upload costs. dotnet run measure.cs -- <cli.dll> <server> <container> <sizeMB> <work>
#:property TargetFramework=net10.0
#:property ManagePackageVersionsCentrally=false
#:property PublishAot=false
using System.Diagnostics;
using System.Net.Http.Headers;
using System.Net.Http.Json;
using System.Text.Json.Nodes;

var (cli, server, container, sizeMb, work) = (args[0], args[1], args[2], int.Parse(args[3]), args[4]);
Directory.CreateDirectory(work);
var http = new HttpClient { BaseAddress = new Uri(server) };
async Task<JsonNode?> Send(HttpMethod m, string path, object? body = null, string? token = null)
{
    using var req = new HttpRequestMessage(m, path);
    if (body is not null) req.Content = JsonContent.Create(body);
    if (token is not null) req.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
    using var res = await http.SendAsync(req);
    var text = await res.Content.ReadAsStringAsync();
    Console.WriteLine($"  {m} {path} → {(int)res.StatusCode}");
    return string.IsNullOrWhiteSpace(text) || !text.TrimStart().StartsWith('{') && !text.TrimStart().StartsWith('[') ? null : JsonNode.Parse(text);
}
async Task<(int, string)> Run(string file, params string[] a)
{
    var psi = new ProcessStartInfo(file) { RedirectStandardOutput = true, RedirectStandardError = true };
    foreach (var x in a) psi.ArgumentList.Add(x);
    using var p = Process.Start(psi)!;
    var o = p.StandardOutput.ReadToEndAsync(); var e = p.StandardError.ReadToEndAsync();
    await p.WaitForExitAsync();
    return (p.ExitCode, await o + await e);
}
var setup = await Send(HttpMethod.Get, "/api/auth/setup");
var session = await Send(HttpMethod.Post, setup?["needsSetup"]?.GetValue<bool>() == true ? "/api/auth/register" : "/api/auth/login", new { username = "admin", password = "admin-password" });
var token = session!["accessToken"]!.GetValue<string>();
var run = DateTime.UtcNow.ToString("HHmmss");
var backend = await Send(HttpMethod.Post, "/api/storage-backends", new { name = $"m {run}", settings = new { type = "Local", path = $"/data/m-{run}" } }, token);
var folder = await Send(HttpMethod.Post, "/api/folders", new { name = $"big{run}", mode = "TwoWay", encryption = "Transport", kind = "Files", storageBackendId = backend!["id"]!.GetValue<int>(), rootPath = "big", intervalMinutes = 60, retention = new { last = 10 } }, token);
var folderId = folder!["id"]!.GetValue<int>();
var config = Path.Combine(work, "config");
var code = (await Send(HttpMethod.Post, "/api/devices/enrollments", new { reusable = false, label = "m" }, token))!["code"]!.GetValue<string>();
Console.WriteLine((await Run("dotnet", cli, "--config", config, "enroll", "--server", server, "--code", code, "--name", "measure")).Item2.Trim());
var deviceId = JsonNode.Parse(File.ReadAllText(Path.Combine(config, "identity.json")))!["deviceId"]!.GetValue<string>();
var local = Path.Combine(work, "big");
Directory.CreateDirectory(local);
await Send(HttpMethod.Put, $"/api/folders/{folderId}/devices", new { deviceId, localPath = local }, token);

void MakeFile(int seed)
{
    var rng = new Random(seed);
    var buf = new byte[1 << 20];
    using var f = File.Create(Path.Combine(local, "big.bin"));
    for (var i = 0; i < sizeMb; i++) { rng.NextBytes(buf); f.Write(buf); }
}

async Task<(double Seconds, double PeakMemMb, double PeakTmpMb, string Output)> Sync(double? killAtFraction = null)
{
    var psi = new ProcessStartInfo("dotnet") { RedirectStandardOutput = true, RedirectStandardError = true };
    foreach (var x in new[] { cli, "--config", config, "sync", "--folder", $"{folderId}" }) psi.ArgumentList.Add(x);
    var sw = Stopwatch.StartNew();
    using var p = Process.Start(psi)!;
    var o = p.StandardOutput.ReadToEndAsync(); var e = p.StandardError.ReadToEndAsync();
    double peakMem = 0, peakTmp = 0;
    while (!p.HasExited)
    {
        var (_, stats) = await Run("docker", "stats", "--no-stream", "--format", "{{.MemUsage}}", container);
        var mem = stats.Split('/')[0].Trim();
        var memMb = mem.EndsWith("GiB") ? double.Parse(mem[..^3]) * 1024 : mem.EndsWith("MiB") ? double.Parse(mem[..^3]) : 0;
        var (_, du) = await Run("docker", "exec", container, "sh", "-c", "du -sm /tmp | cut -f1");
        double.TryParse(du.Trim(), out var tmpMb);
        // Parts already on the storage (new build: hidden staging file next to the target).
        var (_, st) = await Run("docker", "exec", container, "sh", "-c", "find /data -name '*.async-tmp' -exec stat -c %s {} + 2>/dev/null | awk '{s+=$1} END {print int(s/1048576)}'");
        double.TryParse(st.Trim(), out var stagedMb);
        peakMem = Math.Max(peakMem, memMb); peakTmp = Math.Max(peakTmp, tmpMb);
        Console.WriteLine($"    t={sw.Elapsed.TotalSeconds,5:F1}s server mem={memMb,7:F0} MiB  server /tmp={tmpMb,6:F0} MiB  staged={stagedMb,6:F0} MiB");
        if (killAtFraction is double k && Math.Max(tmpMb, stagedMb) >= sizeMb * k) { p.Kill(true); Console.WriteLine("    killed the agent"); break; }
    }
    await p.WaitForExitAsync();
    return (sw.Elapsed.TotalSeconds, peakMem, peakTmp, (await o + await e).Trim());
}

MakeFile(1);
Console.WriteLine($"\n== full upload of {sizeMb} MB ==");
var full = await Sync();
Console.WriteLine($"  {full.Output}\n  RESULT full: {full.Seconds:F1}s, peak server mem {full.PeakMemMb:F0} MiB, peak server /tmp {full.PeakTmpMb:F0} MiB");

MakeFile(2);
Console.WriteLine($"\n== interrupted upload (kill at ~50%) ==");
var cut = await Sync(0.5);
Console.WriteLine($"  RESULT interrupted after {cut.Seconds:F1}s");
var again = await Sync();
Console.WriteLine($"  {again.Output}\n  RESULT retry: {again.Seconds:F1}s, peak server /tmp {again.PeakTmpMb:F0} MiB (a resume would need far less than the full time)");
File.Delete(Path.Combine(local, "big.bin"));
